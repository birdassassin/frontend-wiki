      &amp;amp;#039;Content-Type&amp;amp;#039;: &amp;amp;#039;application/json&amp;amp;#039;,
      ...options?.headers
    }
  });
  
  if (!response.ok) {
    throw new Error(`HTTP ${response.status}`);
  }
  
  return response.json();
}
