## 1. ES6 模块基础

### 1.1 export 导出

```javascript
// math.js

// 导出单个变量
export const PI = 3.14159;

// 导出函数
export function add(a, b) {
  return a + b;
}

// 导出类
export class Calculator {
  multiply(a, b) {
    return a * b;
  }
}

// 导出默认值
export default function subtract(a, b) {
  return a - b;
}

// 批量导出
const divide = (a, b) => a / b;
const square = x => x * x;
export { divide, square };
```

### 1.2 import 导入

```javascript
// main.js

// 导入默认值
import subtract from './math.js';

// 导入命名导出
import { PI, add, Calculator } from './math.js';

// 导入并重命名
import { divide as div, square as sq } from './math.js';

// 导入所有
import * as math from './math.js';

// 使用
console.log(PI);           // 3.14159
console.log(add(1, 2));    // 3
console.log(subtract(5, 3)); // 2
console.log(math.divide(10, 2)); // 5
```

## 2. 模块特性

### 2.1 模块作用域

```javascript
// module.js
const message = 'Hello from module';

export function getMessage() {
  return message;
}

// 在模块外部无法访问 message
```

### 2.2 模块只执行一次

```javascript
// counter.js
let count = 0;

export function increment() {
  return ++count;
}

export function getCount() {
  return count;
}

// main.js
import { increment, getCount } from './counter.js';

console.log(increment()); // 1
console.log(increment()); // 2
console.log(getCount());  // 2
```

## 3. 动态导入

### 3.1 基本用法

```javascript
// 动态导入返回 Promise
import('./math.js').then(math => {
  console.log(math.add(1, 2)); // 3
});

// async/await 方式
async function loadModule() {
  const math = await import('./math.js');
  console.log(math.add(1, 2)); // 3
}
```

### 3.2 条件导入

```javascript
async function loadFeature(feature) {
  if (feature === 'chart') {
    const chart = await import('./chart.js');
    chart.render();
  } else if (feature === 'map') {
    const map = await import('./map.js');
    map.init();
  }
}
```

### 3.3 按需加载（代码分割）

```javascript
// React 中的代码分割
const HeavyComponent = React.lazy(() => import('./HeavyComponent.js'));

function App() {
  return (
    <React.Suspense fallback={<div>Loading...</div>}>
      <HeavyComponent />
    </React.Suspense>
  );
}
```

## 4. 模块路径解析

### 4.1 相对路径

```javascript
import { add } from './math.js';      // 同级目录
import { add } from '../utils/math.js'; // 上级目录
import { add } from './utils/math.js';  // 子目录
```

### 4.2 绝对路径

```javascript
import { add } from '/src/utils/math.js'; // 项目根目录
```

### 4.3 第三方模块

```javascript
import React from 'react';
import { useState } from 'react';
```
