## 1. String.prototype.replaceAll

### 1.1 基本用法

```javascript
const str = 'Hello, World! Hello, JavaScript!';

// ES5 方式（使用全局正则）
const result1 = str.replace(/Hello/g, 'Hi');

// ES2021 replaceAll
const result2 = str.replaceAll('Hello', 'Hi');

console.log(result1); // 'Hi, World! Hi, JavaScript!'
console.log(result2); // 'Hi, World! Hi, JavaScript!'
```

### 1.2 注意事项

```javascript
// replaceAll 要求第一个参数是全局的
const str = 'aaa';

// 如果是正则，必须有 g 标志
str.replaceAll(/a/, 'b'); // ❌ TypeError

// 正确方式
str.replaceAll(/a/g, 'b'); // ✅ 'bbb'
str.replaceAll('a', 'b');  // ✅ 'bbb'
```

## 2. Promise.any

### 2.1 基本用法

```javascript
const p1 = new Promise((resolve) => setTimeout(() => resolve('First'), 300));
const p2 = new Promise((resolve) => setTimeout(() => resolve('Second'), 200));
const p3 = new Promise((resolve) => setTimeout(() => resolve('Third'), 100));

// 返回第一个成功的 Promise
Promise.any([p1, p2, p3])
  .then(result => console.log(result)); // 'Third'
```

### 2.2 全部失败的情况

```javascript
const p1 = Promise.reject('Error 1');
const p2 = Promise.reject('Error 2');
const p3 = Promise.reject('Error 3');

Promise.any([p1, p2, p3])
  .catch(error => {
    console.log(error.errors); // ['Error 1', 'Error 2', 'Error 3']
  });
```

## 3. WeakRef

### 3.1 基本用法

```javascript
const obj = { name: 'Alice' };
const weakRef = new WeakRef(obj);

// 获取引用
console.log(weakRef.deref()?.name); // 'Alice'

// 如果对象被垃圾回收，deref() 返回 undefined
```

### 3.2 应用场景

```javascript
class Cache {
  constructor() {
    this.cache = new Map();
  }
  
  set(key, value) {
    // 使用 WeakRef 避免阻止垃圾回收
    this.cache.set(key, new WeakRef(value));
  }
  
  get(key) {
    const ref = this.cache.get(key);
    return ref?.deref();
  }
}
```

## 4. FinalizationRegistry

### 4.1 基本用法

```javascript
const registry = new FinalizationRegistry((heldValue) => {
  console.log('Object garbage collected:', heldValue);
});

let obj = { name: 'Alice' };

// 注册对象
registry.register(obj, 'obj1');

// 取消引用
obj = null;

// 当 obj 被垃圾回收时，回调会被调用
```

```javascript

const billion = 1_000_000_000;
const million = 1_000_000;
```

## 6. 逻辑赋值运算符

### 6.1 &&= 逻辑与赋值

```javascript
// ES5 方式
if (options) {
  options = options.value;
}

// ES2021
options &&= options.value;
```

### 6.2 ||= 逻辑或赋值

```javascript
// ES5 方式
if (!options.timeout) {
  options.timeout = 5000;
}

// ES2021
options.timeout ||= 5000;
```

### 6.3 ??= 逻辑空赋值

```javascript
// ES5 方式
if (options.timeout === null || options.timeout === undefined) {
  options.timeout = 5000;
}

// ES2021
options.timeout ??= 5000;
```
