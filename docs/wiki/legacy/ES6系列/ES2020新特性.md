## 1. BigInt

### 1.1 基本用法

```javascript
// 创建 BigInt
const bigInt1 = 123n;
const bigInt2 = BigInt(123);

console.log(bigInt1 === bigInt2); // true

// 运算
console.log(10n + 20n); // 30n
console.log(10n * 20n); // 200n
console.log(10n / 3n);  // 3n（整数除法）

// 比较
console.log(10n == 10); // true
console.log(10n === 10); // false（类型不同）

// 转换
console.log(Number(10n)); // 10
console.log(BigInt(10));  // 10n
```

### 1.2 应用场景

```javascript
// 处理超大数字
const maxSafeInteger = Number.MAX_SAFE_INTEGER;
console.log(maxSafeInteger); // 9007199254740991

// 超过安全整数范围
console.log(maxSafeInteger + 1 === maxSafeInteger + 2); // true ❌

// 使用 BigInt
console.log(9007199254740991n + 1n === 9007199254740991n + 2n); // false ✅
```

## 2. Optional Chaining

### 2.1 基本用法

```javascript
const user = {
  name: 'Alice',
  address: {
    city: 'Beijing'
  }
};

// ES5 方式
const city = user && user.address && user.address.city;

// ES2020 Optional Chaining
const city = user?.address?.city;

// 数组访问
const arr = [1, 2, 3];
const first = arr?.[0];

// 函数调用
const fn = null;
const result = fn?.(); // undefined
```

### 2.2 实际应用

```javascript
function getUserCity(user) {
  return user?.address?.city ?? 'Unknown';
}

console.log(getUserCity({ name: 'Alice', address: { city: 'Beijing' } })); // 'Beijing'
console.log(getUserCity({ name: 'Bob' })); // 'Unknown'
console.log(getUserCity(null)); // 'Unknown'
```

## 3. Nullish Coalescing

### 3.1 基本用法

```javascript
// ES5 方式（存在问题）
const count = 0 || 10;
console.log(count); // 10（0 被认为是 falsy）

// ES2020 Nullish Coalescing
const count = 0 ?? 10;
console.log(count); // 0（只有 null/undefined 才使用默认值）

const name = '' ?? 'Guest';
console.log(name); // ''（空字符串不是 nullish）
```

### 3.2 对比 || 和 ??

| 值 | `value || default` | `value ?? default` |
|----|---------------------|---------------------|
| `null` | default | default |
| `undefined` | default | default |
| `0` | default | 0 |
| `''` | default | '' |
| `false` | default | false |
| `NaN` | default | NaN |

## 4. Promise.allSettled

### 4.1 基本用法

```javascript
const p1 = Promise.resolve('Success');
const p2 = Promise.reject('Error');
const p3 = Promise.resolve('Another success');

Promise.allSettled([p1, p2, p3])
  .then(results => {
    results.forEach(result => {
      if (result.status === 'fulfilled') {
        console.log('Success:', result.value);
      } else {
        console.log('Error:', result.reason);
      }
    });
  });

// Output:
// Success: Success
// Error: Error
// Success: Another success
```

## 5. String.prototype.matchAll

### 5.1 基本用法

```javascript
const str = 'Hello 123 World 456';
const regex = /(\w+) (\d+)/g;

// 传统方式
let match;
while ((match = regex.exec(str)) !== null) {
  console.log(match);
}

// ES2020 matchAll
for (const match of str.matchAll(regex)) {
  console.log(match);
}
```

## 6. 动态 import()

### 6.1 基本用法

```javascript
// 动态导入返回 Promise
import('./module.js')
  .then(module => {
    module.doSomething();
  })
  .catch(error => {
    console.error('Failed to load module:', error);
  });

// async/await 方式
async function loadModule() {
  const module = await import('./module.js');
  module.doSomething();
}
```
