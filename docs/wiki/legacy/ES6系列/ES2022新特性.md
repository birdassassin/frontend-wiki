## 1. Class Fields

### 1.1 公共类字段

```javascript
class Person {
  // 公共字段
  name = 'Unknown';
  age = 18;
  
  constructor(name, age) {
    if (name) this.name = name;
    if (age) this.age = age;
  }
}

const person = new Person('Alice', 25);
console.log(person.name); // 'Alice'
console.log(person.age);  // 25
```

### 1.2 私有类字段

```javascript
class Person {
  // 私有字段（# 前缀）
  #password;
  #ssn;
  
  constructor(name, password, ssn) {
    this.name = name;
    this.#password = password;
    this.#ssn = ssn;
  }
  
  // 私有方法
  #validatePassword(password) {
    return this.#password === password;
  }
  
  login(password) {
    return this.#validatePassword(password);
  }
}

const person = new Person('Alice', '123456', '123-45-6789');
console.log(person.#password); // ❌ 报错：私有属性不可访问
console.log(person.login('123456')); // ✅ true
```

### 1.3 静态私有字段

```javascript
class Counter {
  static #count = 0;
  
  constructor() {
    Counter.#count++;
  }
  
  static get total() {
    return Counter.#count;
  }
}

const c1 = new Counter();
const c2 = new Counter();
console.log(Counter.total); // 2
```

## 2. Top-level await

### 2.1 基本用法

```javascript
// module.js
const response = await fetch('/api/data');
const data = await response.json();
export default data;

// main.js
import data from './module.js';
console.log(data);
```

### 2.2 应用场景

```javascript
// 动态加载配置
const config = await import('./config.js');

// 初始化数据库连接
const db = await initDatabase();

// 获取环境变量
const env = await getEnvironment();

export { config, db, env };
```

## 3. Object.hasOwn

### 3.1 基本用法

```javascript
const obj = { name: 'Alice' };
obj.__proto__ = { age: 25 };

// ES5 方式（有问题）
console.log(obj.hasOwnProperty('name')); // true
console.log(obj.hasOwnProperty('age'));  // false

// 如果对象没有 hasOwnProperty 方法？
const badObj = { hasOwnProperty: false };
// badObj.hasOwnProperty('name') // ❌ TypeError

// ES2022 Object.hasOwn
console.log(Object.hasOwn(obj, 'name')); // true
console.log(Object.hasOwn(obj, 'age'));  // false
console.log(Object.hasOwn(badObj, 'name')); // false
```

## 4. Array.prototype.at

### 4.1 基本用法

```javascript
const arr = [1, 2, 3, 4, 5];

// ES5 方式获取最后一个元素
console.log(arr[arr.length - 1]); // 5

// ES2022 at()
console.log(arr.at(-1)); // 5
console.log(arr.at(-2)); // 4
console.log(arr.at(0));  // 1
console.log(arr.at(1));  // 2
```

## 5. RegExp Match Indices

### 5.1 基本用法

```javascript
const str = 'Hello, World!';
const regex = /(\w+), (\w+)/d; // d 标志启用匹配索引

const match = str.match(regex);
console.log(match);
// [
//   'Hello, World',
//   'Hello',
//   'World',
//   indices: [ [0, 12], [0, 5], [7, 12] ]
// ]

// 获取捕获组的位置
console.log(match.indices[1]); // [0, 5]
console.log(match.indices[2]); // [7, 12]
```

## 6. Error Cause

### 6.1 基本用法

```javascript
try {
  await fetch('/api/data');
} catch (error) {
  throw new Error('Failed to fetch data', { cause: error });
}

// 捕获并访问原始错误
try {
  await doSomething();
} catch (error) {
  console.log(error.message); // 'Failed to fetch data'
  console.log(error.cause);   // 原始错误对象
}
```
