# 回文检测，支持字符串和数字类型
```javascript
function isPalindrome(line=&amp;amp;quot;&amp;amp;quot;) {
  for(var i=0,j=line.length-1;i&amp;amp;lt;j;i++,j--){
    // 头尾开始比较是否一致
    if(line.charAt(i) !== line.charAt(j)){
       return false;  // 不是回文
    }
  }
    return true;  //是回文
}

isPalindrome(123321) // true
```
