# 十进制转换2-9进制
```javascript
//原理：取余数，商继续取余数。
function mulBase(num, base) {
  if(typeof num != &amp;amp;#039;number&amp;amp;#039;) throw Error(&amp;amp;#039;num 必须是数字类型&amp;amp;#039;)
  if(base &amp;amp;lt; 2 || base &amp;amp;gt; 9) throw Error(&amp;amp;#039;base 的范围是2-9&amp;amp;#039;)
  var s = []
  do {
    s.push(num % base)
      num = Math.floor(num / base)
  } while(num &amp;amp;gt; 0)
  return Number(s.join(&amp;amp;#039;&amp;amp;#039;))
}

//使用方法
mulBase(10, 2) // 10011
```
